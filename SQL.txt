1. What is SQL?

-- QL (Structured Query Language) is used to:

 -Store data
 -Retrieve data
 - Update/Delete data
  - Manage databases


2. Types of SQL Commands


1. DDL (Data Definition Language)

-CREATE
-ALTER
-DROP
-TRUNCATE


2. DML (Data Manipulation Language)


-INSERT
-UPDATE
-DELETE


3. DQL (Data Query Language)

-SELECT

4. DCL (Data Control Language)

-GRANT
-REVOKE

5. TCL (Transaction Control Language)

-COMMIT
-ROLLBACK
-SAVEPOINT

3. CREATE DATABASE & TABLE

   CREATE DATABASE college;

   USE college;

   CREATE TABLE students (
      id INT PRIMARY KEY,
      name VARCHAR(50),
      age INT,
      marks INT
);


4. INSERT DATA

  INSERT INTO students VALUES (1, 'Shubham', 18, 90);
  INSERT INTO students (id, name, age, marks) VALUES (2, 'Rahul', 19, 85);

5. SELECT DATA

   SELECT * FROM students;
   SELECT name, marks FROM students;
   WHERE Clause
   SELECT * FROM students WHERE marks > 80;


6. OPERATORS

- Comparison

=, >, <, >=, <=, !=

-Logical
  SELECT * FROM students WHERE marks > 80 AND age < 20;
  SELECT * FROM students WHERE marks > 90 OR age < 18;
  SELECT * FROM students WHERE NOT marks < 50;

7. UPDATE
   
   UPDATE students
   SET marks = 95
   WHERE id = 1;

8. DELETE
  
  DELETE FROM students WHERE id = 2;

9. ORDER BY
  
  SELECT * FROM students ORDER BY marks ASC;
  SELECT * FROM students ORDER BY marks DESC;

10. LIMIT

  SELECT * FROM students LIMIT 3;

11. DISTINCT
  
  SELECT DISTINCT age FROM students;

12. AGGREGATE FUNCTIONS

   SELECT COUNT(*) FROM students;
   SELECT SUM(marks) FROM students;
   SELECT AVG(marks) FROM students;
   SELECT MAX(marks) FROM students;
   SELECT MIN(marks) FROM students;

13. GROUP BY
  
  SELECT age, COUNT(*)
  FROM students
  GROUP BY age;

14. HAVING
  
  SELECT age, AVG(marks)
  FROM students
  GROUP BY age
  HAVING AVG(marks) > 80;

15. ALTER TABLE
  
  ALTER TABLE students ADD city VARCHAR(50);
  ALTER TABLE students DROP COLUMN city;

16. DROP vs TRUNCATE

  DROP TABLE students;       -- deletes table completely
  TRUNCATE TABLE students;  -- deletes data only

17. KEYS
  
  Primary Key
  Unique + Not NULL
  Foreign Key
  CREATE TABLE orders (
    order_id INT,
    student_id INT,
    FOREIGN KEY (student_id) REFERENCES students(id)
  );

18. JOINS
   
   INNER JOIN
   SELECT students.name, orders.order_id
   FROM students
   INNER JOIN orders
   ON students.id = orders.student_id;
   LEFT JOIN
   SELECT * FROM students
   LEFT JOIN orders
   ON students.id = orders.student_id;
   RIGHT JOIN
   SELECT * FROM students
   RIGHT JOIN orders
   ON students.id = orders.student_id;


19. SUBQUERIES
 
   SELECT name FROM students
   WHERE marks > (SELECT AVG(marks) FROM students);

20. VIEWS
   
   CREATE VIEW top_students AS
   SELECT name, marks FROM students WHERE marks > 90;
   SELECT * FROM top_students;


21. INDEX

    CREATE INDEX idx_name ON students(name);

22. CONSTRAINTS
 
   NOT NULL
   UNIQUE
   PRIMARY KEY
   FOREIGN KEY
   CHECK
   DEFAULT
   CREATE TABLE users (
    id INT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    age INT CHECK (age >= 18)
  );

23. TRANSACTIONS


  START TRANSACTION;
  UPDATE students SET marks = 100 WHERE id = 1;
  ROLLBACK;   -- undo
  COMMIT;     -- save
